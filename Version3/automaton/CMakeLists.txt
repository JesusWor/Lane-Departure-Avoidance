cmake_minimum_required(VERSION 3.16)
project(lda_automaton CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra)
endif()

# Dependencias: usa las del sistema (sudo apt install libzmq3-dev cppzmq-dev nlohmann-json3-dev);
# si no estan, intenta bajarlas (necesita internet).
find_package(cppzmq QUIET)
find_package(nlohmann_json QUIET)
if(NOT cppzmq_FOUND OR NOT nlohmann_json_FOUND)
  include(FetchContent)
  if(NOT cppzmq_FOUND)
    find_package(ZeroMQ REQUIRED)
    FetchContent_Declare(cppzmq GIT_REPOSITORY https://github.com/zeromq/cppzmq.git GIT_TAG v4.10.0)
    set(CPPZMQ_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(cppzmq)
  endif()
  if(NOT nlohmann_json_FOUND)
    FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
    FetchContent_MakeAvailable(json)
  endif()
endif()

add_executable(lda_automaton src/main.cpp)
target_include_directories(lda_automaton PRIVATE include)
target_link_libraries(lda_automaton PRIVATE cppzmq nlohmann_json::nlohmann_json)

add_executable(test_logic tests/test_logic.cpp)
target_include_directories(test_logic PRIVATE include)
enable_testing()
add_test(NAME logic COMMAND test_logic)
